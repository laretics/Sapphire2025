# HTTP para acceso interno por IP
server {
        #redireccion a HTTPS
        listen 80;
        server_name 88.99.33.109;

        root /home/Zafiro/Client/wwwroot;
        index index.html;

        #Blazor WASM
        location / {
                try_files $uri $uri/ /index.html;
        }

        # API REST
        location /api/ {
                proxy_pass http://localhost:5000/;
                proxy_http_version 1.1;

                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;

                proxy_set_header Upgrade $http_upgrade;
                proxy_set_header Connection keep-alive;
        }
        # TOURMALINE
        location /tourmaline/ {
                proxy_pass http://localhost:5000/api/;
                proxy_http_version 1.1;
                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
                proxy_set_header Upgrade $http_upgrade;
                proxy_set_header Connection "upgrade";
        }
}
# HTTP en dominio para HTTPS dominio (solo para acceso externo)
server {
        listen 80;
        server_name material.trensfm.com www.material.trensfm.com;
        return 301 https://$host$request_uri;
}
# HTTPS para acceso externo por dominio
server{
        listen 443 ssl;
        server_name material.trensfm.com www.material.trensfm.com;

        root /home/Zafiro/Client/wwwroot;
        index index.html index.htm;

        ssl_certificate /etc/ssl/corporate/fullchain.pem;
        ssl_certificate_key /etc/ssl/corporate/certificate.key;

        ssl_protocols TLSv1.2 TLSv1.3;
        ssl_ciphers HIGH:!aNULL:!ND5;

        # Blazor WASM
        location / {
                try_files $uri $uri/ /index.html;
        }

        # API REST de .NET
        location /api/ {
                proxy_pass http://localhost:5000;
                proxy_http_version 1.1;

                proxy_set_header Host $host;
                proxy_set_header X-Real-IP $remote_addr;
                proxy_set_header X-Forwarded_For $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded_Proto $scheme;

                proxy_set_header Upgrade $http_upgrade;
                proxy_set_header Connection keep-alive;
        }
}
